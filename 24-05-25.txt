public class DBConnection {

	public static void main(String[] args) {
		// JDBC
		try{
			Class.forName("com.mysql.cj.jdbc.Driver");
			
			Connection con=DriverManager.getConnection("jdbc:mysql://localhost:3306/vit","root","Armaan#25");
			System.out.println(con);
			Statement stmt=con.createStatement();
			ResultSet rs=stmt.executeQuery("select * from student");
			while(rs.next())
				System.out.println(rs.getInt(1)+" "+rs.getString(2));
			con.close();
		}
			
		
		catch(Exception e){

			e.printStackTrace();
		}

package crud_operation;

import java.sql.Connection;
import java.sql.PreparedStatement;

public class CRUD_Operation {

	public static void main(String[] args) {
		CRUD_Operation objTest=new CRUD_Operation();
		  
		objTest.create_data("107", "manoj");
		objTest.create_data("104", "karthick");
		// TODO Auto-generated method stub

	}
	public void create_data(String sl_no,String name){
		DB_Connection obj_DB_Connection=new DB_Connection();
		Connection connection=obj_DB_Connection.get_connection();
		PreparedStatement ps=null;
		try {
			String query="insert into student values (?,?)";
			ps=connection.prepareStatement(query);
			ps.setString(1, sl_no);
			ps.setString(2, name);
			
			System.out.println(ps);
			ps.executeUpdate();
		} catch (Exception e) {
			System.out.println(e);
		}
	}

}
